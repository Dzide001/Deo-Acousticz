package com.droidacoustic.pro.acoustics

import kotlin.test.*

class DiffuseGeometryWindowsTest {
    @Test fun windowsPreserveSurfaceOrderWithoutExceedingEntryBudget() {
        val entries=intArrayOf(2,3,0,4,1,2)
        val windows=DiffuseGeometryWindows.ranges(entries,5)
        assertEquals(listOf(0..2,3..4,5..5),windows)
        assertEquals(entries.indices.toList(),windows.flatMap {it.toList()})
        assertTrue(windows.all {r -> r.sumOf {entries[it]}<=5})
    }
    @Test fun oversizedCellsAndZeroCapacityKeepTheUncachedFallback() {
        assertEquals(listOf(0..0,1..1,2..2),DiffuseGeometryWindows.ranges(intArrayOf(2,9,2),3))
        assertEquals(listOf(0..0,1..1),DiffuseGeometryWindows.ranges(intArrayOf(1,1),0))
        assertEquals(listOf(0..2),DiffuseGeometryWindows.ranges(intArrayOf(0,0,0),0))
        assertTrue(DiffuseGeometryWindows.ranges(intArrayOf(),0).isEmpty())
    }
    @Test fun rejectsInvalidBudgetsAndCounts() {
        assertFailsWith<IllegalArgumentException> {DiffuseGeometryWindows.ranges(intArrayOf(1),-1)}
        assertFailsWith<IllegalArgumentException> {DiffuseGeometryWindows.ranges(intArrayOf(-1),1)}
    }
}
